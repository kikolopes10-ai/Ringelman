<script>
  function updateDateTime() {
    const now = new Date();
    const p = n => String(n).padStart(2,'0');
    document.getElementById('datetime').textContent =
      `${p(now.getDate())}/${p(now.getMonth()+1)}/${now.getFullYear()} ` +
      `${p(now.getHours())}:${p(now.getMinutes())}:${p(now.getSeconds())}`;
  }
  setInterval(updateDateTime, 1000);
  updateDateTime();

  async function startCamera() {
    const status = document.getElementById('status');
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: { ideal: 'environment' } }, audio: false
      });
      document.getElementById('video').srcObject = stream;
      status.textContent = 'Câmera ativa.';
    } catch(e) {
      status.textContent = 'Erro câmera (necessário HTTPS): ' + e.message;
    }
  }

  function getLocation() {
    const status = document.getElementById('status');
    if (!navigator.geolocation) { status.textContent = 'GPS não suportado.'; return; }
    status.textContent = 'Obtendo localização...';
    navigator.geolocation.getCurrentPosition(
      pos => {
        document.getElementById('lat').textContent = pos.coords.latitude.toFixed(6);
        document.getElementById('lon').textContent = pos.coords.longitude.toFixed(6);
        status.textContent = 'Localização atualizada.';
      },
      err => { status.textContent = 'Erro GPS: ' + err.message; },
      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
    );
  }
  document.getElementById('btnGPS').addEventListener('click', getLocation);

  function buildScale() {
    const wrap = document.getElementById('svgWrap');
    const cont = document.getElementById('container');
    const size = Math.min(cont.clientWidth, cont.clientHeight) * 0.92;
    if (!size) return;

    const ns  = 'http://www.w3.org/2000/svg';
    const cx  = size / 2;
    const cy  = size / 2;
    const R   = size / 2 - 2;      
    const Ro  = R * 0.82;          
    const Ri  = R * 0.45; 

    const N   = 5;
    const startAngle = -Math.PI / 2;

    const svg = document.createElementNS(ns, 'svg' );
    // ID adicionado para captura
    svg.setAttribute('id', 'ringelmannSVG');
    svg.setAttribute('width',   size);
    svg.setAttribute('height',  size);
    svg.setAttribute('viewBox', `0 0 ${size} ${size}`);
    svg.style.background = 'transparent';

    const toXY = (r, a) => [cx + r * Math.cos(a), cy + r * Math.sin(a)];

    const outerRing = document.createElementNS(ns, 'path');
    const dOuter = `M ${cx} ${cy-R} A ${R} ${R} 0 1 1 ${cx} ${cy+R} A ${R} ${R} 0 1 1 ${cx} ${cy-R} Z ` +
                   `M ${cx} ${cy-Ro} A ${Ro} ${Ro} 0 1 0 ${cx} ${cy+Ro} A ${Ro} ${Ro} 0 1 0 ${cx} ${cy-Ro} Z`;
    outerRing.setAttribute('d', dOuter);
    outerRing.setAttribute('fill', '#ffffff');
    outerRing.setAttribute('stroke', '#000000');
    outerRing.setAttribute('stroke-width', 1.5);
    svg.appendChild(outerRing);

    const segs = [
      { color: '#d8d8d8', label: 'Nº1', dens: '20%'  }, 
      { color: '#aaaaaa', label: 'Nº2', dens: '40%'  }, 
      { color: '#787878', label: 'Nº3', dens: '60%'  }, 
      { color: '#484848', label: 'Nº4', dens: '80%'  }, 
      { color: '#000000', label: 'Nº5', dens: '100%' }  
    ];

    const step = (2 * Math.PI) / N;
    segs.forEach((seg, i) => {
      const a0 = startAngle + i * step;
      const a1 = startAngle + (i + 1) * step;
      const [ox0, oy0] = toXY(Ro, a0);
      const [ox1, oy1] = toXY(Ro, a1);
      const [ix1, iy1] = toXY(Ri, a1);
      const [ix0, iy0] = toXY(Ri, a0);

      const d = `M ${ox0} ${oy0} A ${Ro} ${Ro} 0 0 1 ${ox1} ${oy1} L ${ix1} ${iy1} L ${ix0} ${iy0} Z`;
      const path = document.createElementNS(ns, 'path');
      path.setAttribute('d', d);
      path.setAttribute('fill', seg.color);
      path.setAttribute('stroke', '#ffffff');
      path.setAttribute('stroke-width', 1.2);
      svg.appendChild(path);
    });

    const rText = (Ro + R) / 2;
    segs.forEach((seg, i) => {
      const a0 = startAngle + i * step;
      const a1 = startAngle + (i + 1) * step;
      const pathId = `textArc${i}`;
      const arcPath = document.createElementNS(ns, 'path');
      const [sx, sy] = toXY(rText, a0 + 0.08);
      const [ex, ey] = toXY(rText, a1 - 0.08);
      arcPath.setAttribute('id', pathId);
      arcPath.setAttribute('d', `M ${sx} ${sy} A ${rText} ${rText} 0 0 1 ${ex} ${ey}`);
      arcPath.setAttribute('fill', 'none');
      svg.appendChild(arcPath);

      const textEl = document.createElementNS(ns, 'text');
      textEl.setAttribute('font-size', size * 0.032);
      textEl.setAttribute('font-family', 'Arial, sans-serif');
      textEl.setAttribute('font-weight', 'bold');
      textEl.setAttribute('fill', '#000000');
      
      const tp = document.createElementNS(ns, 'textPath');
      tp.setAttribute('href', `#${pathId}`);
      tp.setAttribute('startOffset', '50%');
      tp.setAttribute('text-anchor', 'middle');
      tp.textContent = `${seg.label} DENS. ${seg.dens}`;
      
      textEl.appendChild(tp);
      svg.appendChild(textEl);
    });

    for (let i = 0; i < N; i++) {
      const a = startAngle + i * step;
      const [x1, y1] = toXY(Ro, a); const [x2, y2] = toXY(R, a);
      const sep = document.createElementNS(ns, 'line');
      sep.setAttribute('x1', x1); sep.setAttribute('y1', y1);
      sep.setAttribute('x2', x2); sep.setAttribute('y2', y2);
      sep.setAttribute('stroke', '#000000'); sep.setAttribute('stroke-width', 1.5);
      svg.appendChild(sep);
    }

    wrap.innerHTML = '';
    wrap.appendChild(svg);
  }

  // NOVA FUNÇÃO DE CAPTURA
  function capturePhoto() {
    const video = document.getElementById('video');
    const canvas = document.getElementById('canvasCapture');
    const ctx = canvas.getContext('2d');
    const svg = document.getElementById('ringelmannSVG');
    
    // 1. Ajusta o canvas para a resolução real do vídeo
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    
    // 2. Desenha o frame do vídeo no canvas
    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
    
    // 3. Serializa o SVG da sua escala para imagem
    const svgData = new XMLSerializer().serializeToString(svg);
    const svgBlob = new Blob([svgData], {type: 'image/svg+xml;charset=utf-8'});
    const url = URL.createObjectURL(svgBlob);
    const img = new Image();

    img.onload = function() {
      // Desenha o seu SVG centralizado sobre a foto
      const size = Math.min(canvas.width, canvas.height) * 0.9;
      const x = (canvas.width - size) / 2;
      const y = (canvas.height - size) / 2;
      ctx.drawImage(img, x, y, size, size);
      
      // 4. Adiciona texto com data e GPS na foto
      ctx.fillStyle = "white";
      ctx.strokeStyle = "black";
      ctx.lineWidth = 3;
      ctx.font = "20px Arial";
      const info = `${document.getElementById('datetime').textContent} | LAT: ${document.getElementById('lat').textContent} LON: ${document.getElementById('lon').textContent}`;
      ctx.strokeText(info, 20, canvas.height - 30);
      ctx.fillText(info, 20, canvas.height - 30);

      // 5. Gera o download
      const link = document.createElement('a');
      link.download = `Ringelmann_${Date.now()}.png`;
      link.href = canvas.toDataURL("image/png");
      link.click();
      URL.revokeObjectURL(url);
    };
    img.src = url;
  }
  document.getElementById('btnCapture').addEventListener('click', capturePhoto);

  window.addEventListener('resize', buildScale);
  startCamera();
  getLocation();
  setTimeout(buildScale, 100);
</script>
